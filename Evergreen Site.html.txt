            function calculateTotal() {
                let total = 0;
                let daysSelected = 0;
                let earlyCount = 0;
                let lateCount = 0;

                const selectedDays = [];

                dayCheckboxes.forEach(checkbox => {
                    if (checkbox.checked) {
                        const price = parseInt(checkbox.dataset.price || '72', 10);
                        total += price;
                        daysSelected++;
                        selectedDays.push(checkbox.closest('[data-day]').dataset.day);
                    }
                });

                addonCheckboxes.forEach(checkbox => {
                    const day = checkbox.dataset.day;
                    if (selectedDays.includes(day) && checkbox.checked) {
                        const price = parseInt(checkbox.dataset.price, 10);
                        total += price;
                        if (checkbox.dataset.type === 'early') earlyCount++;
                        if (checkbox.dataset.type === 'late') lateCount++;
                    }
                });

                totalPriceEl.textContent = '£' + total;

                const summaryParts = [];
                if (daysSelected > 0) summaryParts.push(daysSelected + ' day' + (daysSelected !== 1 ? 's' : ''));
                if (earlyCount > 0) summaryParts.push(earlyCount + ' early');
                if (lateCount > 0) summaryParts.push(lateCount + ' late');
                summaryTextEl.textContent = summaryParts.length > 0 ? summaryParts.join(' + ') + ' selected' : 'No days selected';

                const netTotal = (total / 1.2).toFixed(2);
                const vatAmount = (total - parseFloat(netTotal)).toFixed(2);
                priceBreakdownEl.textContent = '£' + netTotal + ' + £' + vatAmount + ' VAT';

                addToCartBtn.disabled = daysSelected === 0;

                dayCheckboxes.forEach(dayCheckbox => {

                        addons.forEach(addon => {
                            addon.disabled = true;
                            addon.checked = false;
                        });
                    }
                });
            }

            function getCookie(name) {
                const nameEQ = name + "=";
                const ca = document.cookie.split(';');
                for (let i = 0; i < ca.length; i++) {
                    let c = ca[i];
                    while (c.charAt(0) === ' ') c = c.substring(1, c.length);
                    if (c.indexOf(nameEQ) === 0) return decodeURIComponent(c.substring(nameEQ.length, c.length));
                }
                return null;
            }

            function getSquarespaceCrumb() {
                try {
                    if (window.Static && window.Static.SQUARESPACE_CONTEXT && window.Static.SQUARESPACE_CONTEXT.crumb) {
                        return window.Static.SQUARESPACE_CONTEXT.crumb;
                    }
                } catch (e) {}

                const cookieCrumb = getCookie('crumb');
                if (cookieCrumb) return cookieCrumb;

                return null;
            }

            async function addOneItemToSquarespaceCart(product) {
                const crumb = getSquarespaceCrumb();
                if (!crumb) {
                    const err = new Error('Missing Squarespace crumb. Cookie "crumb" not found and context crumb not available.');
                    err.code = 'NO_CRUMB';
                    throw err;
                }

                const url = `/api/commerce/shopping-cart/entries?crumb=${encodeURIComponent(crumb)}`;

                const payload = {
                    itemId: product.itemId,
                    sku: product.sku,
                    quantity: 1,
                    additionalFields: "null"
                };

                const res = await fetch(url, {
                    method: 'POST',
                    credentials: 'same-origin',
                    headers: {
                        'Content-Type': 'application/json',
                        'Accept': 'application/json'
                    },
                    body: JSON.stringify(payload)
                });

                const text = await res.text();
                let data = null;
                try { data = text ? JSON.parse(text) : null; } catch (e) { data = text; }

                if (!res.ok) {
                    console.group('Squarespace cart add failed');
                    console.log('Status:', res.status, res.statusText);
                    console.log('URL:', url);
                    console.log('Product:', product);
                    console.log('Payload:', payload);
                    console.log('Response:', data);
                    console.groupEnd();

                    const err = new Error(`Cart request failed: ${res.status} ${res.statusText}`);
                    err.status = res.status;
                    err.response = data;
                    throw err;
                }

                return data;
            }

            function handleAddToCart() {

                const selectedItems = [];

                dayCheckboxes.forEach(checkbox => {
                    if (!checkbox.checked) return;

                    const dayKey = checkbox.closest('[data-day]').dataset.day;
                    const config = DAY_CONFIG[dayKey];

                    // Add day
                    selectedItems.push({
                        name: config.name,
                        price: 72,
                        url: config.dayUrl
                    });

                    // Check for early
                    const earlyBox = document.querySelector(`.addon-checkbox[data-day="${dayKey}"][data-type="early"]`);
                    if (earlyBox && earlyBox.checked) {
                        selectedItems.push({
                            name: `Early Drop-off - ${config.name}`,
                            price: 12,
                            url: config.earlyUrl
                        });
                    }

                    // Check for late
                    const lateBox = document.querySelector(`.addon-checkbox[data-day="${dayKey}"][data-type="late"]`);
                    if (lateBox && lateBox.checked) {
                        selectedItems.push({
                            name: `Late Pick-up - ${config.name}`,
                            price: 18,
                            url: config.lateUrl
                        });
                    }

                                    Close
                                </button>
